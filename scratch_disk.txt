<script src="/assets/js/mv-shell.js"></script>
<script>
(function () {
    'use strict';

    /* Doctor Dashboard — HERO is prescription builder with medicine autocomplete */

    let schedule = [];
    let selectedAppt = null;
    let rxItems = [];
    let medicines = [];
    let consultModalInst;

    (async function init() {
        const ctx = await MV.shell.mount({
            roles: ['Doctor'],
            title: 'My Schedule',
            subtitle: 'Today'
        });

        ctx.content.innerHTML =
            '<div class="schedule-stats" id="scheduleStats"></div>' +
            '<div class="timeline" id="scheduleTimeline"></div>' +
            '<div class="mv-footer">' +
              'MediVault — Hospital Management System | DBMS Project by Kris Johnson • Roll No. 2024XXXX' +
            '</div>';

        consultModalInst = new bootstrap.Modal(document.getElementById('consultModal'));

        document.getElementById('saveConsultation').addEventListener('click', handleSaveConsultation);

        loadSchedule();
        loadMedicines();
    })();

    async function loadSchedule() {
        const stats = document.getElementById('scheduleStats');
        const timeline = document.getElementById('scheduleTimeline');

        stats.innerHTML = Array(3).fill('<div class="mv-stat"><div class="mv-skeleton" style="width:40px;height:28px;margin:0 auto 8px"></div><div class="mv-skeleton" style="width:60px;height:12px;margin:0 auto"></div></div>').join('');
        timeline.innerHTML = Array(3).fill('<div class="timeline-item"><div class="mv-skeleton" style="width:50px;height:40px"></div><div style="flex:1"><div class="mv-skeleton" style="width:120px;height:14px;margin-bottom:8px"></div><div class="mv-skeleton" style="width:80px;height:12px"></div></div></div>').join('');

        try {
            schedule = await MV.api.get('/api/doctor/schedule');

            const today = new Date().toISOString().slice(0, 10);
            const todayCount = schedule.filter(function (a) { return a.appointment_date && a.appointment_date.slice(0, 10) === today; }).length;
            const upcoming = schedule.filter(function (a) { return a.appointment_date && a.appointment_date.slice(0, 10) > today; }).length;
            const completed = schedule.filter(function (a) { return a.status === 'Completed'; }).length;

            stats.innerHTML =
                '<div class="mv-stat"><div class="mv-stat-value">' + todayCount + '</div><div class="mv-stat-label">Today</div></div>' +
                '<div class="mv-stat"><div class="mv-stat-value">' + upcoming + '</div><div class="mv-stat-label">Upcoming</div></div>' +
                '<div class="mv-stat"><div class="mv-stat-value">' + completed + '</div><div class="mv-stat-label">Completed</div></div>';

            if (!schedule.length) {
                timeline.innerHTML = '<div class="mv-empty"><div class="mv-empty-icon"><i class="fas fa-calendar"></i></div>' +
                    '<p class="mv-empty-title">No appointments today or upcoming</p></div>';
                return;
            }

            // Group by date
            const byDate = {};
            schedule.forEach(function (appt) {
                const d = appt.appointment_date ? appt.appointment_date.slice(0, 10) : 'Unknown';
                if (!byDate[d]) byDate[d] = [];
                byDate[d].push(appt);
            });

            let html = '';
            Object.keys(byDate).sort().forEach(function (dateKey) {
                const isToday = dateKey === today;
                const dateLabel = isToday ? 'Today — ' + MV.fmt.dateIndian(dateKey) : MV.fmt.dateIndian(dateKey);
                html += '<div style="font-size:12px;font-weight:600;color:var(--mv-text-muted);text-transform:uppercase;letter-spacing:0.5px;margin:16px 0 8px;padding:0 4px">' + MV.esc(dateLabel) + '</div>';
                byDate[dateKey].forEach(function (appt) {
                    const statusBadge = appt.status === 'Completed' ? '<span class="mv-badge is-success">Done</span>' :
                        appt.status === 'Cancelled' ? '<span class="mv-badge is-danger">Cancelled</span>' :
                        '<span class="mv-badge is-info">Waiting</span>';
                    html += '<div class="timeline-item" onclick="openConsultation(' + appt.appointment_id + ')">' +
                        '<div class="timeline-time">' + MV.fmt.time12(appt.appointment_time) + '</div>' +
                        '<div class="timeline-content">' +
                          '<h5>' + MV.esc(appt.patient_name) + '</h5>' +
                          '<p>' + statusBadge + '</p>' +
                        '</div>' +
                      '</div>';
                });
            });
            timeline.innerHTML = html;

        } catch (err) {
            timeline.innerHTML = '<div class="mv-empty"><p class="mv-empty-text">Failed to load schedule</p></div>';
        }
    }

    async function loadMedicines() {
        try {
            medicines = await MV.api.get('/api/doctor/medicines/search?q=');
        } catch (err) {
            medicines = [];
        }
    }

    window.openConsultation = async function (appointmentId) {
        selectedAppt = schedule.find(function (a) { return a.appointment_id === appointmentId; });
        if (!selectedAppt) return;

        rxItems = [{ medicine_id: null, medicine_name: '', dosage: '', duration_days: '', quantity: '' }];

        const body = document.getElementById('consultBody');
        body.innerHTML = '<div class="text-center py-4"><div class="spinner-border text-primary"></div></div>';
        consultModalInst.show();

        try {
            const history = await MV.api.get('/api/doctor/patients/' + selectedAppt.patient_id + '/history');

            body.innerHTML =
                '<div style="margin-bottom:24px">' +
                  '<h6 style="font-size:16px;font-weight:600;margin:0 0 8px">Patient: ' + MV.esc(selectedAppt.patient_name) + '</h6>' +
                  '<p style="font-size:12px;color:var(--mv-text-muted);margin:0">Appointment: ' + MV.fmt.time12(selectedAppt.appointment_time) + '</p>' +
                '</div>' +

                '<section class="rx-history"><h6>Clinical history</h6>' +
                  (history.records.length ? history.records.map(r=>'<p>'+MV.fmt.date(r.created_at)+' · '+MV.esc(r.diagnosis_summary)+'</p>').join('') : '<p class="text-muted">No earlier medical records.</p>') +
                '</section>' +
                '<form id="consultForm">' +
                  '<div class="mb-3">' +
                    '<label class="form-label">Symptoms</label>' +
                    '<textarea class="form-control" name="symptoms" rows="2" required></textarea>' +
                  '</div>' +
                  '<div class="mb-3">' +
                    '<label class="form-label">Diagnosis</label>' +
                    '<textarea class="form-control" name="diagnosis" rows="2" required></textarea>' +
                  '</div>' +
                  '<div class="mb-3">' +
                    '<label class="form-label">Notes</label>' +
                    '<textarea class="form-control" name="notes" rows="2"></textarea>' +
                  '</div>' +
                '</form>' +

                '<div class="rx-builder">' +
                  '<h6>Prescription</h6>' +
                  '<div id="rxContainer"></div>' +
                  '<button type="button" class="mv-btn mv-btn-outline mv-btn-sm rx-add-btn" onclick="addRxItem()">' +
                    '<i class="fas fa-plus"></i> Add Medicine' +
                  '</button>' +
                '</div>';

            renderRxItems();

        } catch (err) {
            body.innerHTML = '<p class="text-danger">Failed to load patient data</p>';
        }
    };

    function renderRxItems() {
        const container = document.getElementById('rxContainer');
        if (!container) return;

        container.innerHTML =
            '<div class="table-responsive">' +
              '<table class="table table-borderless align-middle mb-2" style="font-size:13px">' +
                '<thead><tr class="text-muted" style="font-size:11.5px;text-transform:uppercase;letter-spacing:0.5px">' +
                  '<th style="width:36%;padding:4px 8px">Medicine</th>' +
                  '<th style="width:22%;padding:4px 8px">Dosage</th>' +
                  '<th style="width:18%;padding:4px 8px">Duration (Days)</th>' +
                  '<th style="width:18%;padding:4px 8px">Quantity</th>' +
                  '<th style="width:6%;padding:4px 8px"></th>' +
                '</tr></thead>' +
                '<tbody>' +
                rxItems.map(function (item, idx) {
                    return '<tr>' +
                        '<td style="padding:4px 8px">' +
                          '<input type="text" class="form-control form-control-sm" list="medicineList" placeholder="Search medicine..." ' +
                            'value="' + MV.esc(item.medicine_name) + '" ' +
                            'onchange="updateRxMedicine(' + idx + ', this.value)">' +
                        '</td>' +
                        '<td style="padding:4px 8px">' +
                          '<input type="text" class="form-control form-control-sm" placeholder="e.g. 500mg" ' +
                            'value="' + MV.esc(item.dosage) + '" ' +
                            'onchange="updateRxField(' + idx + ', \'dosage\', this.value)">' +
                        '</td>' +
                        '<td style="padding:4px 8px">' +
                          '<input type="number" class="form-control form-control-sm" placeholder="7" ' +
                            'value="' + (item.duration_days || '') + '" ' +
                            'onchange="updateRxField(' + idx + ', \'duration_days\', this.value)">' +
                        '</td>' +
                        '<td style="padding:4px 8px">' +
                          '<input type="number" class="form-control form-control-sm" placeholder="14" ' +
                            'value="' + (item.quantity || '') + '" ' +
                            'onchange="updateRxField(' + idx + ', \'quantity\', this.value)">' +
                        '</td>' +
                        '<td style="padding:4px 8px;text-align:center">' +
                          (rxItems.length > 1 ? '<button type="button" class="mv-btn mv-btn-ghost mv-btn-icon mv-btn-sm text-danger" title="Remove" onclick="removeRxItem(' + idx + ')">' +
                            '<i class="fas fa-trash-alt"></i></button>' : '') +
                        '</td>' +
                      '</tr>';
                }).join('') +
                '</tbody>' +
              '</table>' +
            '</div>' +
            '<datalist id="medicineList">' +
              medicines.map(function (m) {
                  return '<option value="' + MV.esc(m.medicine_name) + '">';
              }).join('') +
            '</datalist>';
    }

    window.addRxItem = function () {
        rxItems.push({ medicine_id: null, medicine_name: '', dosage: '', duration_days: '', quantity: '' });
        renderRxItems();
    };

    window.removeRxItem = function (idx) {
        rxItems.splice(idx, 1);
        renderRxItems();
    };

    window.updateRxMedicine = function (idx, name) {
        const med = medicines.find(function (m) { return m.medicine_name === name; });
        rxItems[idx].medicine_name = name;
        rxItems[idx].medicine_id = med ? med.medicine_id : null;
    };

    window.updateRxField = function (idx, field, value) {
        rxItems[idx][field] = value;
    };

    async function handleSaveConsultation() {
        const form = document.getElementById('consultForm');
        if (!form || !form.checkValidity()) {
            if (form) form.reportValidity();
            return;
        }

        const formData = Object.fromEntries(new FormData(form));
        const btn = document.getElementById('saveConsultation');
        const restore = MV.pending(btn, 'Saving...');

        try {
            // Save consultation
            const consultRes = await MV.api.post('/api/doctor/consultations', {
                appointment_id: selectedAppt.appointment_id,
                symptoms: formData.symptoms,
                diagnosis: formData.diagnosis,
                notes: formData.notes
            });

            // Save prescription if any items filled
            const validItems = rxItems.filter(function (item) {
                return item.medicine_id && item.dosage && item.duration_days && item.quantity;
            });

            if (validItems.length > 0) {
                await MV.api.post('/api/doctor/prescriptions', {
                    consultation_id: consultRes.consultation_id,
                    patient_id: selectedAppt.patient_id,
                    items: validItems
                });
            }

            MV.toast.success('Consultation saved');
            consultModalInst.hide();
            loadSchedule();

        } catch (err) {
            MV.toast.error('Save failed', err.message);
        } finally {
            restore();
        }
    }
})();
</script>
</body>
